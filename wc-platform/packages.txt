# apt packages for Streamlit Community Cloud.
#
# libreoffice-writer, not the full libreoffice suite: it provides the
# same `soffice` binary used for docx -> PDF at a fraction of the
# install size, which matters because the full suite frequently exceeds
# the Community Cloud resource limit and fails the build. If even this
# fails to install, the app degrades to shipping the .docx alone rather
# than erroring - see pipeline/pdf_converter.PdfUnavailable.
libreoffice-writer

# cairosvg loads libcairo.so.2 at import time. Without this the mind map
# fails at import, not at render.
libcairo2

# The rupee glyph. Arial substitutes a missing box for it, which is the
# reason every SVG text node specifies DejaVu Sans explicitly.
fonts-dejavu
