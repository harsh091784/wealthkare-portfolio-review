# Pinned exactly. Streamlit Community Cloud installs from this file on
# every rebuild, so an unpinned dependency means the app can start
# failing on a day nobody touched it.
#
# Kept at the REPOSITORY ROOT alongside packages.txt. Streamlit Cloud
# would also read this one from the entrypoint's directory, but
# packages.txt is root-only, and splitting the two across directories is
# how packages.txt came to be silently ignored on the first deploy.
#
# Every pin below has a prebuilt cp313 wheel. matplotlib 3.11.1 has no
# cp314 wheel, so on Python 3.14 pip falls back to building it from
# source - deploy the app on Python 3.13.

streamlit==1.61.1
openpyxl==3.1.5            # .xlsx reading (read_only + data_only)
python-docx==1.2.0         # report construction
matplotlib==3.11.1         # donut / gauge / scatter charts
cairosvg==2.9.0            # mind map SVG -> PNG (needs libcairo2, see packages.txt)
pymupdf==1.28.2            # TOC page detection, PDF page counts
pillow==12.3.0             # font metrics for column widths; imported directly by
                           # docx_builder, mindmap and chart_gen, so pinned in its
                           # own right rather than relied on as a matplotlib
                           # transitive dependency
anthropic==0.122.0         # the single Client Summary call
